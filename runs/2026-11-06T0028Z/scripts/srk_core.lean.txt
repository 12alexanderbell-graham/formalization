lemma ratio_acentric {r : ℝ} (h1 : (1018337 / 100000000) ≤ r) (h2 : r ≤ (1018689 / 100000000)) :
    RoundsTo r 0.0102 4 ∧ RoundsTo (-Real.logb 10 r - 1) 0.992 3 := by
  have hr : 0 < r := by linarith
  refine ⟨?_, ?_⟩
  · unfold RoundsTo; rw [abs_le]; constructor <;> norm_num <;> linarith
  · have a1 := logR_lo
    have a2 := logR_hi
    have a3 := log1000_lo
    have a4 := log1000_hi
    have m1 : Real.log (1018337 / 100000000) ≤ Real.log r := Real.log_le_log (by norm_num) h1
    have m2 : Real.log r ≤ Real.log (1018689 / 100000000) := Real.log_le_log hr h2
    have e : Real.log (1000 : ℝ) = 3 * Real.log 10 := by
      rw [show (1000 : ℝ) = 10 ^ 3 by norm_num, Real.log_pow]; norm_num
    have hL0 : 0 < Real.log 10 := Real.log_pos (by norm_num)
    have A : -19925 / 10000 * Real.log 10 ≤ Real.log r := by linarith
    have B : Real.log r ≤ -19915 / 10000 * Real.log 10 := by linarith
    have A' : -19925 / 10000 ≤ Real.log r / Real.log 10 := (le_div_iff₀ hL0).2 A
    have B' : Real.log r / Real.log 10 ≤ -19915 / 10000 := (div_le_iff₀ hL0).2 B
    unfold RoundsTo Real.logb
    generalize Real.log r / Real.log 10 = t at A' B' ⊢
    rw [abs_le]
    constructor <;> norm_num <;> linarith

lemma alg_cancel (A B K P : ℝ) (hB : B ≠ 0) (hK : K ≠ 0) (hP : P ≠ 0) :
    A * K ^ 2 / P / (B * K / P) ^ 2 / P = A / B ^ 2 := by
  field_simp

  -- α = (1 + κ(1)(1 − √0.7))²; then p_s/P_c = π · α · (x/9)/(x²/3 − x − 1)² and ω from `ratio_acentric`.
  have hD : 0 < vc ^ 2 + 1 * b * vc + 0 * b ^ 2 := by nlinarith

    (by linear_combination hf) (by norm_num) (by norm_num) (by norm_num) (by norm_num)

  have e2 : ((x - 1) * (2 * x + 1) - (x ^ 2 + 1 * x + 0)) / ((x - 1) ^ 2 * (2 * x + 1))
      = x ^ 2 / 3 - x - 1 := by
    rw [div_eq_iff (by positivity)]; linear_combination (-2 * x ^ 2 / 3 + x) * hf
  have e3 : (x ^ 2 + 1 * x + 0) ^ 2 * ((x - 1) * (2 * x + 1) - (x ^ 2 + 1 * x + 0)) /
      ((x - 1) ^ 4 * (2 * x + 1) ^ 2) = x / 9 := by
    rw [div_eq_iff (by positivity)]
    linear_combination (-4 * x ^ 4 / 9 + x ^ 3 + 2 * x ^ 2 / 3 + x / 9) * hf

    rw [h0, div_zero] at h2
    linarith
  have hR0 : R ≠ 0 := left_ne_zero_of_mul hRT
  have hTc0 : Tc ≠ 0 := right_ne_zero_of_mul hRT

    have : 9 / (384732211 / 100000000) ≤ 9 / x :=
      div_le_div_of_nonneg_left (by norm_num) (by linarith) hi.le
    linarith
  have hKpos : 0 ≤ 3 * x - 9 - 9 / x := by
    have : (0 : ℝ) ≤ 3 * (38473221 / 10000000) - 9 - 9 / (38473221 / 10000000) := by norm_num
